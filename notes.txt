export AMENT_PREFIX_PATH=$HOME/ros2_ws/install/annin_ar4_description:$AMENT_PREFIX_PATH

export GAZEBO_MODEL_PATH=$GAZEBO_MODEL_PATH:~/ros2_ws/install/annin_ar4_description/share/annin_ar4_description/meshes


Launch AR4 in Rvis
ros2 launch annin_ar4_moveit_config demo.launch.py ar_model:=mk4

ros2 launch annin_ar4_gazebo gazebo.launch.py ar_model:=mk5
