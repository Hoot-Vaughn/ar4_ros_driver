how to manually import packages to ros2: (not needed with bare metal install-needed for WSL2)
export AMENT_PREFIX_PATH=$HOME/ros2_ws/install/annin_ar4_description:$AMENT_PREFIX_PATH
export AMENT_PREFIX_PATH=$HOME/ros2_ws/install/annin_ar4_driver:$AMENT_PREFIX_PATH
export AMENT_PREFIX_PATH=$HOME/ros2_ws/install/annin_ar4_firmware:$AMENT_PREFIX_PATH
export AMENT_PREFIX_PATH=$HOME/ros2_ws/install/annin_ar4_gazebo:$AMENT_PREFIX_PATH
export AMENT_PREFIX_PATH=$HOME/ros2_ws/install/annin_ar4_moveit_config:$AMENT_PREFIX_PATH
sudo adduser $USER dialout



Gazebo (how to fix gazebo to load the ar4 arm)
export GZ_SIM_RESOURCE_PATH=~/ros2_ws/install/annin_ar4_description/share:~/ros2_ws/src/ar4_ros_driver
export GZ_MODEL_PATH=~/ros2_ws/install/annin_ar4_description/share:~/ros2_ws/src/ar4_ros_driver

Automate 
echo 'export GZ_SIM_RESOURCE_PATH=~/ros2_ws/install/annin_ar4_description/share:~/ros2_ws/src/ar4_ros_driver' >> ~/.bashrc

Launch AR4 in Rvis
ros2 launch annin_ar4_moveit_config demo.launch.py ar_model:=mk4

Launch AR4 in Rvis for Gazebo
ros2 launch annin_ar4_moveit_config moveit.launch.py use_sim_time:=true include_gripper:=True



Launch Gazebo
ros2 launch annin_ar4_gazebo gazebo.launch.py ar_model:=mk4
